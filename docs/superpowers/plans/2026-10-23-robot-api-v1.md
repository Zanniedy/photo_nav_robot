# 机器人本地 /api/v1 与任务层 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 机器人有自己的 robot_id，对外提供统一的 `/api/v1`（token 或 cookie 认证、统一错误码），能下发 goto / guide / patrol 任务并拿到进度和结果，巡航可等待上层交互后继续，支持软件急停。

**Architecture:** web 后端（FastAPI）里加一个纯 Python 任务状态机 `task_runner.py`，通过 `NavPort` 接口驱动导航；生产环境由 `ros_link.py`（后端进程内的 rclpy 节点）实现 `NavPort`，测试用 `FakeNav`。`runtime.py` 持有单例，`routers/v1_*.py` 提供接口，旧接口改调任务层。急停在 `robot_bridge` 里落地（`/robot/estop`）。

**Tech Stack:** Python 3.10（机器人和本机 `/usr/bin/python3`）、FastAPI 0.111、rclpy（ROS 2 Humble）、nav2_msgs、pytest、C++17 + gtest（robot_bridge）、Vue 3（网页一处小改）。

**Spec:** `docs/superpowers/specs/2026-10-23-robot-api-v1-design.md`

## Global Constraints

- 机器人身份文件 `~/max_robot/robot.yaml`，**不进仓库**（`rsync.sh --delete` 会覆盖 `ros_ws/src` 下的文件）
- `robot_id` 只允许 `[a-z0-9-]`，1~32 字符；已配置后不能再改（返回 `ROBOT_NOT_CONFIGURED`）
- `/api/v1` 成功 `{"ok": true, "robot_id": ..., "data": ...}`，失败 `{"ok": false, "robot_id": ..., "code": ..., "msg": ...}`
- 错误码固定为：`UNAUTHORIZED`、`BAD_REQUEST`、`NOT_FOUND`、`AMBIGUOUS`、`NAV_NOT_RUNNING`、`LOCALIZATION_UNRELIABLE`、`MAP_MISMATCH`、`ESTOPPED`、`BUSY`、`NAV_FAILED`、`ROBOT_NOT_CONFIGURED`、`INTERNAL`
- 对外地图标识 `map_id` = 地图名，`version` = yaml + 图片内容 sha1 前 12 位
- 默认值：guide 限速 0.15 m/s；`wait_on_person` 关；`person_distance` 3m；`wait_timeout` 120s（0 = 一直等）；`person_cooldown` 30s；`min_score` 0（不检查）；任务历史保留 50 个
- 旧接口（`/api/points_goto`、`/api/nav/waypoints`、`/api/nav/cancel`）保持请求/返回格式不变
- 代码风格跟现有后端一致：中文注释、模块顶部 docstring 说明用途、不加多余抽象
- 本机跑 Python 测试**只用** `ros_ws/src/web/backend/tests/run.sh`（干净环境 + ROS + `.venv_web`），不要用 conda 的 python
- 本机编译 ROS 包用干净环境：`env -i HOME=$HOME PATH=/usr/bin:/bin bash -c 'source /opt/ros/humble/setup.bash && cd ros_ws && colcon build --packages-select <包>'`

```python
"""
机器人身份：robot.yaml（config.yaml 的 robot_file，默认 ~/max_robot/robot.yaml），每台机器人一份，不进仓库。
  robot_id: "max-001"            全局唯一，[a-z0-9-]，上云后 MQTT 账号、topic 都用它，配置后不能再改
  name: "一号机"                  显示名，随时可改
  api_tokens: [{name, token}]    /api/v1 的 Bearer token（平板、上层调用方）
文件不存在或坏了：robot_id = unset-<hostname>，configured = False，后端照常启动。
"""
import contextlib, copy, fcntl, hmac, os, re, secrets, socket
import yaml
from config import ROBOT_FILE

API_VERSION = "1.0"
MAX_TOKENS = 20
_ID_RE = re.compile(r"^[a-z0-9-]{1,32}$")
_TOKEN_NAME_RE = re.compile(r"^[A-Za-z0-9_\-]{1,32}$")
_cache: dict = {"key": None, "data": None}     # 按文件 mtime/size 缓存，status 每秒都要读 robot_id


class ConfigError(ValueError):
    def __init__(self, code: str, msg: str):
        super().__init__(msg)
        self.code = code


@contextlib.contextmanager
def _locked():
    os.makedirs(os.path.dirname(ROBOT_FILE) or ".", exist_ok=True)
    with open(ROBOT_FILE + ".lock", "w") as lf:
        fcntl.flock(lf, fcntl.LOCK_EX)
        try:
            yield
        finally:
            fcntl.flock(lf, fcntl.LOCK_UN)


def _default() -> dict:
    host = re.sub(r"[^a-z0-9-]", "-", socket.gethostname().lower())[:24] or "robot"
    return {"robot_id": f"unset-{host}", "name": "", "configured": False, "api_tokens": []}


def _read() -> dict:
    try:
        st = os.stat(ROBOT_FILE)
    except FileNotFoundError:
        return _default()
    key = (st.st_mtime_ns, st.st_size)
    if _cache["key"] != key:
        try:
            with open(ROBOT_FILE) as f:
                raw = yaml.safe_load(f) or {}
            if not isinstance(raw, dict):
                raise ValueError
        except (OSError, ValueError, yaml.YAMLError):
            return _default()                       # 文件坏了按没配置处理，不让后端起不来
        d = _default()
        rid = str(raw.get("robot_id") or "")
        if _ID_RE.match(rid):
            d.update(robot_id=rid, configured=True)
        d["name"] = str(raw.get("name") or "")
        d["api_tokens"] = [{"name": str(t["name"]), "token": str(t["token"])}
                           for t in raw.get("api_tokens") or []
                           if isinstance(t, dict) and t.get("name") and t.get("token")]
        _cache.update(key=key, data=d)
    return copy.deepcopy(_cache["data"])


def _write(d: dict) -> None:
    out = {"name": d["name"], "api_tokens": d["api_tokens"]}
    if d["configured"]:
        out = {"robot_id": d["robot_id"], **out}
    tmp = ROBOT_FILE + ".tmp"
    with open(tmp, "w") as f:
        yaml.safe_dump(out, f, allow_unicode=True, sort_keys=False)
    os.chmod(tmp, 0o600)                            # 里面有 token
    os.replace(tmp, ROBOT_FILE)


def _public(d: dict) -> dict:
    return {"robot_id": d["robot_id"], "name": d["name"], "configured": d["configured"],
            "api_version": API_VERSION, "tokens": [t["name"] for t in d["api_tokens"]]}


def load() -> dict:
    with _locked():
        return _read()


def robot_id() -> str:
    return load()["robot_id"]


def public_info() -> dict:
    return _public(load())


def set_identity(robot_id: str | None = None, name: str | None = None) -> dict:
    with _locked():
        d = _read()
        if robot_id is not None and not (d["configured"] and robot_id == d["robot_id"]):
            if d["configured"]:
                raise ConfigError("ROBOT_NOT_CONFIGURED",
                                  "robot_id 已配置，不能再改（确实要改请 ssh 上去编辑 robot.yaml）")
            if not _ID_RE.match(robot_id):
                raise ConfigError("BAD_REQUEST", "robot_id 只能用小写字母、数字和 -，1~32 个字符")
            d.update(robot_id=robot_id, configured=True)
        if name is not None:
            name = name.strip()
            if len(name) > 64:
                raise ConfigError("BAD_REQUEST", "名字最多 64 个字符")
            d["name"] = name
        _write(d)
        return _public(d)


def create_token(name: str) -> str:
    """生成 token，只在这里返回一次明文"""
    if not _TOKEN_NAME_RE.match(name or ""):
        raise ConfigError("BAD_REQUEST", "token 名只能用字母、数字、_ 和 -，1~32 个字符")
    with _locked():
        d = _read()
        if any(t["name"] == name for t in d["api_tokens"]):
            raise ConfigError("BAD_REQUEST", f"token 名已存在: {name}")
        if len(d["api_tokens"]) >= MAX_TOKENS:
            raise ConfigError("BAD_REQUEST", f"token 最多 {MAX_TOKENS} 个")
        tok = secrets.token_urlsafe(32)
        d["api_tokens"].append({"name": name, "token": tok})
        _write(d)
        return tok


def delete_token(name: str) -> bool:
    with _locked():
        d = _read()
        n = len(d["api_tokens"])
        d["api_tokens"] = [t for t in d["api_tokens"] if t["name"] != name]
        if len(d["api_tokens"]) == n:
            return False
        _write(d)
        return True


def check_token(token: str) -> str | None:
    """token 对得上返回它的名字（日志里记是谁调的），否则 None"""
    for t in load()["api_tokens"]:
        if hmac.compare_digest(t["token"].encode(), token.encode()):
            return t["name"]
    return None
```

```python
"""
/api/v1 统一返回格式：
  成功 {"ok": true,  "robot_id": ..., "data": ...}
  失败 {"ok": false, "robot_id": ..., "code": "...", "msg": "..."}
HTTP 状态码按 code 定（见 STATUS），调用方以 code 为准。旧 /api 接口不受影响。
"""
from fastapi import FastAPI, Request
from fastapi.exception_handlers import request_validation_exception_handler
from fastapi.exceptions import RequestValidationError
from fastapi.responses import JSONResponse
import robot_config

STATUS = {
    "UNAUTHORIZED": 401, "BAD_REQUEST": 400, "NOT_FOUND": 404, "AMBIGUOUS": 409,
    "NAV_NOT_RUNNING": 409, "LOCALIZATION_UNRELIABLE": 409, "MAP_MISMATCH": 409,
    "ESTOPPED": 409, "BUSY": 409, "NAV_FAILED": 502, "ROBOT_NOT_CONFIGURED": 409, "INTERNAL": 500,
}


class ApiError(Exception):
    def __init__(self, code: str, msg: str):
        super().__init__(msg)
        self.code = code if code in STATUS else "INTERNAL"
        self.msg = msg

    @property
    def status(self) -> int:
        return STATUS[self.code]


def ok(data=None) -> dict:
    return {"ok": True, "robot_id": robot_config.robot_id(), "data": data}


def fail_body(code: str, msg: str) -> dict:
    return {"ok": False, "robot_id": robot_config.robot_id(), "code": code, "msg": msg}


def fail_response(code: str, msg: str) -> JSONResponse:
    return JSONResponse(fail_body(code, msg), status_code=STATUS.get(code, 500))


def install(app: FastAPI) -> None:
    @app.exception_handler(ApiError)
    async def _api_error(_req: Request, e: ApiError):
        return fail_response(e.code, e.msg)

    @app.exception_handler(RequestValidationError)
    async def _validation(req: Request, e: RequestValidationError):
        if not req.url.path.startswith("/api/v1/"):
            return await request_validation_exception_handler(req, e)   # 旧接口保持 422
        msg = "; ".join(f"{'.'.join(str(x) for x in err['loc'][1:])}: {err['msg']}" for err in e.errors())
        return fail_response("BAD_REQUEST", msg or "参数错误")
```

```python
_version_cache: dict = {}     # (yaml 路径, yaml 的 mtime/size, 图片路径, 图片的 mtime/size) → version


def name_of(path: str) -> str:
    """yaml 路径 → 地图名（对外 map_id）；旧目录格式 <name>/map.yaml 用目录名"""
    base = os.path.basename(path).rsplit(".", 1)[0]
    return os.path.basename(os.path.dirname(path)) if base == "map" else base


def _stat(p: str | None):
    try:
        s = os.stat(p)
        return (s.st_mtime_ns, s.st_size)
    except (OSError, TypeError):
        return None


def _image_path(yaml_path: str) -> str | None:
    try:
        with open(yaml_path) as f:
            img = (yaml.safe_load(f) or {}).get("image")
    except Exception:
        return None
    if not img:
        return None
    return img if os.path.isabs(img) else os.path.join(os.path.dirname(yaml_path), img)


def version(yaml_path: str) -> str:
    """yaml + 图片内容 sha1 前 12 位。按文件 mtime/size 缓存，status 每秒都要算"""
    img = _image_path(yaml_path)
    key = (yaml_path, _stat(yaml_path), img, _stat(img))
    v = _version_cache.get(key)
    if v is None:
        h = hashlib.sha1()
        for p in (yaml_path, img):
            if not p:
                continue
            try:
                with open(p, "rb") as f:
                    for chunk in iter(lambda: f.read(1 << 20), b""):
                        h.update(chunk)
            except OSError:
                pass
        v = h.hexdigest()[:12]
        if len(_version_cache) > 200:
            _version_cache.clear()
        _version_cache[key] = v
    return v
```

```python
def _route_point_ids(d: dict, keys: list[str]) -> list[str]:
    """路线里的点：id 或点名 → id，必须都存在"""
    if not keys:
        raise StoreError("路线至少要一个点")
    if len(keys) > MAX_ROUTE_POINTS:
        raise StoreError(f"路线最多 {MAX_ROUTE_POINTS} 个点")
    ids = []
    for k in keys:
        p = next((p for p in d["points"] if p["id"] == k or p["name"] == k), None)
        if p is None:
            raise StoreError(f"点不存在: {k}")
        ids.append(p["id"])
    return ids


def _find_route(d: dict, key: str) -> dict | None:
    return next((r for r in d["routes"] if r["id"] == key or r["name"] == key), None)


def list_routes(map_id: str) -> list[dict]:
    return load(map_id)["routes"]


def get_route(map_id: str, key: str) -> dict | None:
    return _find_route(load(map_id), key)


def create_route(map_id: str, name: str, points: list[str], loop: bool = True) -> dict:
    if not _NAME_RE.match(name or "") or name.strip() != name:
        raise StoreError("路线名 1~64 字符，不能含 / \\ 或首尾空格")
    with _locked():
        d = _read(map_id)
        if any(r["name"] == name for r in d["routes"]):
            raise StoreError(f"路线名已存在: {name}")
        if len(d["routes"]) >= MAX_ROUTES:
            raise StoreError(f"路线最多 {MAX_ROUTES} 条")
        r = {"id": secrets.token_hex(4), "name": name, "points": _route_point_ids(d, points), "loop": bool(loop)}
        d["routes"].append(r)
        _write(d)
        return r


def update_route(map_id: str, key: str, name: str | None = None, points: list[str] | None = None,
                 loop: bool | None = None) -> dict:
    with _locked():
        d = _read(map_id)
        r = _find_route(d, key)
        if r is None:
            raise KeyError(key)
        if name is not None and name != r["name"]:
            if not _NAME_RE.match(name) or name.strip() != name:
                raise StoreError("路线名 1~64 字符，不能含 / \\ 或首尾空格")
            if any(q["name"] == name for q in d["routes"]):
                raise StoreError(f"路线名已存在: {name}")
            r["name"] = name
        if points is not None:
            r["points"] = _route_point_ids(d, points)
        if loop is not None:
            r["loop"] = bool(loop)
        _write(d)
        return r


def delete_route(map_id: str, key: str) -> bool:

        if len(d["routes"]) == n:
            return False
        _write(d)
```

```python
def pt(name: str, x: float = 0.0, y: float = 0.0, yaw: float = 0.0) -> dict:
    return {"id": f"id-{name}", "name": name, "x": x, "y": y, "yaw": yaw}


class FakeClock:
    def __init__(self):
        self.t = 1000.0

    def __call__(self) -> float:
        return self.t

    def advance(self, dt: float) -> None:
        self.t += dt


class FakeNav:
    """task_runner.NavPort 的替身：记录收到的目标，测试里手动 arrive() / fail()"""
    def __init__(self):
        self.running, self.is_mapping, self.map_id = True, False, "office"
        self.score: float | None = 0.9
        self.robot_pose: tuple | None = (0.0, 0.0, 0.0)
        self.person: float | None = None
        self.goals: list[tuple] = []
        self.callback = None
        self.cancels = 0
        self.speed: float | None = None
        self.estop = False

    def nav_running(self): return self.running
    def mapping(self): return self.is_mapping
    def current_map_id(self): return self.map_id
    def loc_score(self): return self.score
    def pose(self): return self.robot_pose
    def nearest_person(self): return self.person

    def send_goal(self, x, y, yaw, on_done):
        self.goals.append((x, y, yaw))
        self.callback = on_done

    def cancel_goal(self):
        self.cancels += 1

    def set_speed_limit(self, v): self.speed = v
    def set_estop(self, on): self.estop = on

    def arrive(self):
        cb, self.callback = self.callback, None
        x, y, yaw = self.goals[-1]
        self.robot_pose = (x, y, yaw)
        cb(True, "到达")

    def fail(self, msg: str = "导航失败"):
        cb, self.callback = self.callback, None
        cb(False, msg)
```

```python
"""
任务层：goto / guide / patrol 状态机。纯 Python，不依赖 ROS；导航通过 NavPort 驱动
（生产环境 ros_link.RosLink，测试 tests/fakes.FakeNav）。

  goto    依次走到一个或多个点
  guide   带路：限速的单点 goto（默认 guide_speed）
  patrol  按路线巡航，可循环。多一个 waiting 状态：
            moving ──发现人 / wait()──▶ waiting ──resume() / 超时──▶ moving（去下一个没到的点）
          waiting 期间可以执行 goto / guide 子任务，做完仍在 waiting，等上层 resume

同一时间最多一个前台任务（goto / guide）和一个巡航：新前台任务顶掉旧前台任务，但只把巡航挂起；
新巡航顶掉旧巡航和它的子任务。
导航回调可能来自 ROS 线程，状态都在 self._lock 下改；取消 / 换目标后晚到的旧回调靠 _gen 丢掉。
"""
import math, secrets, threading, time
from collections import OrderedDict
from dataclasses import dataclass, field, fields
from typing import Callable, Protocol

HISTORY = 50                        # 保留最近多少个任务（含已结束的）
TERMINAL = ("succeeded", "failed", "canceled", "preempted")


class TaskError(Exception):
    def __init__(self, code: str, msg: str):
        super().__init__(msg)
        self.code = code
        self.msg = msg


class NavPort(Protocol):
    def nav_running(self) -> bool: ...
    def mapping(self) -> bool: ...
    def current_map_id(self) -> str | None: ...
    def loc_score(self) -> float | None: ...
    def pose(self) -> tuple[float, float, float] | None: ...
    def nearest_person(self) -> float | None: ...
    def send_goal(self, x: float, y: float, yaw: float, on_done: Callable[[bool, str], None]) -> None: ...
    def cancel_goal(self) -> None: ...
    def set_speed_limit(self, v: float | None) -> None: ...
    def set_estop(self, on: bool) -> None: ...


@dataclass
class RunnerConfig:
    default_speed: float = 0.25     # 估算剩余时间用
    guide_speed: float = 0.15       # guide 默认限速
    min_score: float = 0.0          # 定位分数门限，0 = 不检查
    wait_on_person: bool = False    # 巡航中发现人自动进入 waiting
    person_distance: float = 3.0
    wait_timeout: float = 120.0     # 0 = 一直等
    person_cooldown: float = 30.0   # resume 后多久内不因为发现人再停

    @classmethod
    def from_dict(cls, d: dict) -> "RunnerConfig":
        names = {f.name for f in fields(cls)}
        return cls(**{k: (bool(v) if k == "wait_on_person" else float(v))
                      for k, v in (d or {}).items() if k in names})


@dataclass
class Task:
    id: str
    type: str                        # goto / guide / patrol
    points: list[dict]               # 下发时的点位快照 {id, name, x, y, yaw}
    map_id: str | None
    max_speed: float | None = None
    loop: bool = False
    route_id: str | None = None
    parent: str | None = None        # 巡航中下发的子任务：所属巡航 id
    state: str = "pending"
    index: int = 0                   # 正在去（或 resume 后要去）的点
    laps: int = 0
    fails: int = 0                   # 巡航连续去不了的点数
    skipped: list[str] = field(default_factory=list)
    message: str = ""
    created: float = 0.0             # 墙钟时间，给人看
    updated: float = 0.0
    wait_since: float | None = None  # clock() 时间，算超时
    wait_reason: str = ""            # manual / person / subtask


class TaskRunner:
    def __init__(self, nav: NavPort, cfg: RunnerConfig | None = None,
                 emit: Callable[[dict], None] = lambda ev: None,
                 clock: Callable[[], float] = time.monotonic, estopped: bool = False):
        self.nav = nav
        self.cfg = cfg or RunnerConfig()
        self.estopped = estopped
        self._emit = emit
        self._clock = clock
        self._lock = threading.RLock()
        self._fg: Task | None = None          # goto / guide
        self._patrol: Task | None = None
        self._driving: Task | None = None     # 当前导航目标属于哪个任务
        self._gen = 0                         # 每发 / 撤一次目标 +1，旧回调对不上就丢
        self._cooldown_until = 0.0
        self._history: OrderedDict[str, Task] = OrderedDict()

    # ── 对外 ────────────────────────────────────────────────────────────

    def check(self, map_id: str | None = None) -> None:
        """下发 / 恢复前的安全检查，不通过抛 TaskError"""
        if self.estopped:
            raise TaskError("ESTOPPED", "急停中，先解除急停")
        if self.nav.mapping():
            raise TaskError("BUSY", "建图模式下不能下发任务")
        if not self.nav.nav_running():
            raise TaskError("NAV_NOT_RUNNING", "导航未启动")
        cur = self.nav.current_map_id()
        if map_id and map_id != cur:
            raise TaskError("MAP_MISMATCH", f"当前地图是 {cur}，不是 {map_id}")
        if self.cfg.min_score > 0:
            s = self.nav.loc_score()
            if s is None or s < self.cfg.min_score:
                raise TaskError("LOCALIZATION_UNRELIABLE",
                                f"定位不可靠（得分 {s}，门限 {self.cfg.min_score}）")

    def submit(self, type: str, points: list[dict], map_id: str | None = None,
               max_speed: float | None = None, loop: bool = False, route_id: str | None = None) -> dict:
        if type not in ("goto", "guide", "patrol"):
            raise TaskError("BAD_REQUEST", f"未知任务类型: {type}")
        if not points:
            raise TaskError("BAD_REQUEST", "没有要去的点")
        if type == "guide":
            if len(points) != 1:
                raise TaskError("BAD_REQUEST", "guide 只能去一个点")
            max_speed = max_speed or self.cfg.guide_speed
        with self._lock:
            self.check(map_id)
            now = time.time()
            t = Task(id=secrets.token_hex(4), type=type, points=[dict(p) for p in points],
                     map_id=self.nav.current_map_id(), max_speed=max_speed,
                     loop=bool(loop) and type == "patrol", route_id=route_id, created=now, updated=now)
            if self._fg is not None:
                self._finish(self._fg, "preempted", "被新任务顶掉")
            if type == "patrol":
                if self._patrol is not None:
                    self._finish(self._patrol, "preempted", "被新的巡航顶掉")
                self._patrol = t
            else:
                if self._patrol is not None:
                    self._suspend(self._patrol, "subtask")
                    t.parent = self._patrol.id
                self._fg = t
            self._remember(t)
            self._set_state(t, "running")
            self._drive(t)
            return self._summary(t)

    def cancel(self, task_id: str) -> dict:
        with self._lock:
            t = self._active(task_id)
            self._finish(t, "canceled", "已取消")
            return self._summary(t)

    def wait(self, task_id: str) -> dict:
        with self._lock:
            t = self._active_patrol(task_id)
            self._suspend(t, "manual")
            return self._summary(t)

    def resume(self, task_id: str) -> dict:
        with self._lock:
            t = self._active_patrol(task_id)
            if t.state != "waiting":
                return self._summary(t)
            if self._fg is not None:
                raise TaskError("BUSY", "子任务还在执行，等它结束或先取消")
            self.check()
            self._resume(t)
            return self._summary(t)

    def get(self, task_id: str) -> dict | None:
        with self._lock:
            t = self._history.get(task_id)
            return self._summary(t) if t else None

    def current(self) -> dict:
        with self._lock:
            return {"foreground": self._summary(self._fg) if self._fg else None,
                    "patrol": self._summary(self._patrol) if self._patrol else None}

    def estop(self) -> None:
        with self._lock:
            self.estopped = True
            self.nav.set_estop(True)
            self.cancel_all("急停")
            self._emit({"type": "estop", "on": True})

    def release_estop(self) -> None:
        with self._lock:
            self.estopped = False
            self.nav.set_estop(False)
            self._emit({"type": "estop", "on": False})

    def cancel_all(self, reason: str) -> None:
        self._end_all("canceled", reason)

    def tick(self) -> None:
        """外部每 0.5s 调一次：导航停了就结束任务、等待超时、巡航中发现人"""
        with self._lock:
            if (self._fg or self._patrol) and not self.nav.nav_running():
                self._end_all("failed", "导航已停止")
                return
            p = self._patrol
            if p is None:
                return
            now = self._clock()
            if p.state == "waiting":
                if self._fg is None and self.cfg.wait_timeout > 0 and now - p.wait_since >= self.cfg.wait_timeout:
                    try:
                        self.check()
                    except TaskError as e:
                        p.wait_since = now                 # 过一个超时周期再试
                        p.message = f"等待超时但不能继续: {e.msg}"
                        return
                    self._emit_task("wait_timeout", p)
                    self._resume(p)
            elif self._driving is p and self.cfg.wait_on_person and now >= self._cooldown_until:
                d = self.nav.nearest_person()
                if d is not None and d <= self.cfg.person_distance:
                    self._emit_task("person_found", p, distance=round(d, 2))
                    self._suspend(p, "person")

    # ── 内部（调用方已持有 self._lock）──────────────────────────────────

    def _end_all(self, state: str, reason: str) -> None:
        with self._lock:
            for t in (self._fg, self._patrol):     # 先前台再巡航
                if t is not None and t.state not in TERMINAL:
                    self._finish(t, state, reason)

    def _active(self, task_id: str) -> Task:
        for t in (self._fg, self._patrol):
            if t is not None and t.id == task_id:
                return t
        old = self._history.get(task_id)
        if old is not None:
            raise TaskError("BAD_REQUEST", f"任务已结束（{old.state}）")
        raise TaskError("NOT_FOUND", f"任务不存在: {task_id}")

    def _active_patrol(self, task_id: str) -> Task:
        t = self._active(task_id)
        if t.type != "patrol":
            raise TaskError("BAD_REQUEST", "只有巡航任务能 wait / resume")
        return t

    def _remember(self, t: Task) -> None:
        self._history[t.id] = t
        while len(self._history) > HISTORY:
            old = next((k for k, v in self._history.items() if v.state in TERMINAL), None)
            if old is None:
                break
            del self._history[old]

    def _set_state(self, t: Task, state: str) -> None:
        t.state = state
        t.updated = time.time()
        self._emit_task("task_state", t)

    def _emit_task(self, type: str, t: Task, **extra) -> None:
        self._emit({"type": type, "task": self._summary(t), **extra})

    def _stop_nav(self) -> None:
        self._gen += 1
        self._driving = None
        self.nav.cancel_goal()
        self.nav.set_speed_limit(None)

    def _drive(self, t: Task) -> None:
        """让 t 去它的第 index 个点"""
        p = t.points[t.index]
        self._gen += 1
        g = self._gen
        self._driving = t
        self.nav.set_speed_limit(t.max_speed)
        self.nav.send_goal(float(p["x"]), float(p["y"]), float(p.get("yaw", 0.0)),
                           lambda ok, msg: self._on_leg_done(g, ok, msg))

    def _suspend(self, p: Task, reason: str) -> None:
        if p.state == "waiting":
            return
        if self._driving is p:
            self._stop_nav()
        p.state = "waiting"
        p.wait_since = self._clock()
        p.wait_reason = reason
        p.updated = time.time()
        self._emit_task("waiting", p, reason=reason)

    def _resume(self, p: Task) -> None:
        p.state = "running"
        p.wait_since = None
        p.wait_reason = ""
        p.updated = time.time()
        self._cooldown_until = self._clock() + self.cfg.person_cooldown
        self._emit_task("resumed", p)
        self._drive(p)

    def _on_leg_done(self, g: int, ok: bool, msg: str) -> None:
        with self._lock:
            if g != self._gen or self._driving is None:
                return                                   # 已取消 / 已换目标的旧回调
            t = self._driving
            self._driving = None
            self.nav.set_speed_limit(None)
            name = t.points[t.index]["name"]
            if ok:
                t.fails = 0
                t.message = f"已到达 {name}"
            elif t.type == "patrol":
                # 巡航里某个点去不了：跳过；连续一整圈都去不了才算失败
                t.fails += 1
                t.skipped = (t.skipped + [name])[-20:]
                if t.fails >= len(t.points):
                    self._finish(t, "failed", f"巡航失败：连续 {t.fails} 个点都去不了（{msg}）")
                    return
                t.message = f"去 {name} 失败，跳过（{msg}）"
            else:
                self._finish(t, "failed", f"去 {name} 失败: {msg}")
                return
            t.index += 1
            if t.index >= len(t.points):
                if not t.loop:
                    self._finish(t, "succeeded", t.message if ok else f"巡航结束，{t.message}")
                    return
                t.index = 0
                t.laps += 1
            t.updated = time.time()
            self._emit_task("progress", t)
            self._drive(t)

    def _finish(self, t: Task, state: str, msg: str) -> None:
        if self._driving is t:
            self._stop_nav()
        t.state = state
        t.message = msg
        t.wait_since = None
        t.updated = time.time()
        if t is self._fg:
            self._fg = None
            if self._patrol is not None and t.parent == self._patrol.id:
                self._patrol.wait_since = self._clock()   # 子任务结束后重新计等待超时
        if t is self._patrol:
            self._patrol = None
            if self._fg is not None and self._fg.parent == t.id:
                self._finish(self._fg, state, f"所属巡航已结束: {msg}")
        self._emit_task("task_state", t)

    def _summary(self, t: Task) -> dict:
        active = t.state in ("running", "waiting")
        target = t.points[t.index] if active and t.index < len(t.points) else None
        remaining = None
        pose = self.nav.pose() if target else None
        if target and pose:
            remaining = math.hypot(target["x"] - pose[0], target["y"] - pose[1])
            rest = t.points[t.index:]
            remaining += sum(math.hypot(b["x"] - a["x"], b["y"] - a["y"]) for a, b in zip(rest, rest[1:]))
        speed = t.max_speed or self.cfg.default_speed
        wait_left = None
        if t.state == "waiting" and self.cfg.wait_timeout > 0 and self._fg is None and t.wait_since is not None:
            wait_left = round(max(0.0, self.cfg.wait_timeout - (self._clock() - t.wait_since)), 1)
        return {
            "id": t.id, "type": t.type, "state": t.state, "map_id": t.map_id,
            "points": [p["name"] for p in t.points], "index": t.index,
            "target": target["name"] if target else None,
            "loop": t.loop, "laps": t.laps, "route_id": t.route_id, "parent": t.parent,
            "max_speed": t.max_speed, "message": t.message, "skipped": list(t.skipped),
            "remaining_m": round(remaining, 2) if remaining is not None else None,
            "eta_s": round(remaining / speed, 1) if remaining is not None else None,
            "wait_reason": t.wait_reason or None if t.state == "waiting" else None,
            "wait_left_s": wait_left,
            "created": t.created, "updated": t.updated,
        }
```

```python
"""事件推送：任务层（任意线程）→ 各个 /api/v1/events WebSocket（asyncio）。慢的客户端丢最旧的事件"""
import asyncio, threading


def _put(q: asyncio.Queue, ev: dict) -> None:
    if q.full():
        q.get_nowait()
    q.put_nowait(ev)


class EventHub:
    def __init__(self, maxsize: int = 200):
        self._subs: set[tuple] = set()
        self._lock = threading.Lock()
        self._maxsize = maxsize

    def subscribe(self) -> tuple:
        """在 asyncio 协程里调用；返回 (loop, queue)，退出时 unsubscribe"""
        sub = (asyncio.get_running_loop(), asyncio.Queue(self._maxsize))
        with self._lock:
            self._subs.add(sub)
        return sub

    def unsubscribe(self, sub: tuple) -> None:
        with self._lock:
            self._subs.discard(sub)

    def publish(self, ev: dict) -> None:
        with self._lock:
            subs = list(self._subs)
        for loop, q in subs:
            try:
                loop.call_soon_threadsafe(_put, q, ev)
            except RuntimeError:          # 事件循环已关闭
                self.unsubscribe((loop, q))
```

- [ ] **Step 5: 写 `web/backend/ros_link.py`**

```python
"""
后端进程内的 rclpy 节点，实现 task_runner.NavPort：
  导航  /navigate_to_pose（nav2_msgs/action/NavigateToPose）
  位姿  TF map → base_footprint
  定位  /localizer/score（Float32），3s 没更新算没有
  人员  /person/state（person_tracker 发的 JSON），1s 没更新算没有
  限速  /speed_limit（nav2_msgs/SpeedLimit，controller_server 订阅；0 = 不限速）
  急停  /robot/estop（Bool，transient local，robot_bridge 订阅）
导航是否启动、当前地图由后端进程管理（process_mgr / stack），构造时以函数传进来。
"""
import json, math, threading, time
import rclpy
import tf2_ros
from action_msgs.msg import GoalStatus
from nav2_msgs.action import NavigateToPose
from nav2_msgs.msg import SpeedLimit
from rclpy.action import ActionClient
from rclpy.executors import MultiThreadedExecutor
from rclpy.node import Node
from rclpy.qos import DurabilityPolicy, QoSProfile
from rclpy.time import Time
from std_msgs.msg import Bool, Float32, String

SCORE_STALE = 3.0
PERSON_STALE = 1.0
_RESULT_TEXT = {
    GoalStatus.STATUS_SUCCEEDED: "到达",
    GoalStatus.STATUS_ABORTED: "导航失败（nav2 中止）",
    GoalStatus.STATUS_CANCELED: "已取消",
}


def person_distance(state: dict, pose) -> float | None:
    """/person/state → 离机器人最近的人的距离（轨迹要在 map 系）"""
    if pose is None or state.get("frame", "map") != "map":
        return None
    ds = [math.hypot(t["x"] - pose[0], t["y"] - pose[1])
          for t in state.get("tracks") or [] if "x" in t and "y" in t]
    return min(ds) if ds else None


class RosLink:
    def __init__(self, nav_running, mapping, current_map_id):
        self.nav_running = nav_running
        self.mapping = mapping
        self.current_map_id = current_map_id
        if not rclpy.ok():
            rclpy.init()
        self.node = Node("web_task_link")
        self._nav = ActionClient(self.node, NavigateToPose, "/navigate_to_pose")
        self._tf = tf2_ros.Buffer()
        self._tf_listener = tf2_ros.TransformListener(self._tf, self.node)
        self._lock = threading.Lock()
        self._score: tuple = (None, 0.0)       # (值, monotonic 时间)
        self._person: tuple = (None, 0.0)
        self._handle = None                    # 当前 goal handle
        self._seq = 0                          # 每发 / 撤一次目标 +1
        self.node.create_subscription(Float32, "/localizer/score", self._on_score, 10)
        self.node.create_subscription(String, "/person/state", self._on_person, 10)
        latched = QoSProfile(depth=1, durability=DurabilityPolicy.TRANSIENT_LOCAL)
        self._estop_pub = self.node.create_publisher(Bool, "/robot/estop", latched)
        self._speed_pub = self.node.create_publisher(SpeedLimit, "/speed_limit", 10)
        self._exec = MultiThreadedExecutor(num_threads=2)
        self._exec.add_node(self.node)
        threading.Thread(target=self._exec.spin, daemon=True).start()

    def shutdown(self) -> None:
        self._exec.shutdown()
        self.node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()

    def _on_score(self, msg: Float32) -> None:
        self._score = (float(msg.data), time.monotonic())

    def _on_person(self, msg: String) -> None:
        try:
            self._person = (json.loads(msg.data), time.monotonic())
        except ValueError:
            pass

    def loc_score(self) -> float | None:
        v, t = self._score
        return v if v is not None and time.monotonic() - t < SCORE_STALE else None

    def pose(self):
        try:
            tf = self._tf.lookup_transform("map", "base_footprint", Time())
        except tf2_ros.TransformException:
            return None
        tr, q = tf.transform.translation, tf.transform.rotation
        return (tr.x, tr.y, math.atan2(2 * (q.w * q.z + q.x * q.y), 1 - 2 * (q.y * q.y + q.z * q.z)))

    def nearest_person(self) -> float | None:
        st, t = self._person
        if st is None or time.monotonic() - t > PERSON_STALE:
            return None
        return person_distance(st, self.pose())

    def send_goal(self, x: float, y: float, yaw: float, on_done) -> None:
        """异步发目标，结果经 on_done(ok, msg) 回调（executor 线程）。被 cancel_goal 撤掉的目标也会回调，
        task_runner 自己丢掉过期回调"""
        if not self._nav.server_is_ready():
            on_done(False, "nav2 action server 不在线")
            return
        goal = NavigateToPose.Goal()
        goal.pose.header.frame_id = "map"            # stamp 留 0 = 用最新 TF
        goal.pose.pose.position.x, goal.pose.pose.position.y = float(x), float(y)
        goal.pose.pose.orientation.z, goal.pose.pose.orientation.w = math.sin(yaw / 2), math.cos(yaw / 2)
        with self._lock:
            self._seq += 1
            seq = self._seq
            self._handle = None

        def on_result(fut):
            try:
                st = fut.result().status
            except Exception as e:
                on_done(False, f"取导航结果失败: {e}")
                return
            on_done(st == GoalStatus.STATUS_SUCCEEDED, _RESULT_TEXT.get(st, f"导航结束 status={st}"))

        def on_accept(fut):
            try:
                h = fut.result()
            except Exception as e:
                on_done(False, f"发送目标失败: {e}")
                return
            if not h.accepted:
                on_done(False, "nav2 拒绝了目标")
                return
            with self._lock:
                if seq != self._seq:                 # 等接受的时候已经被撤 / 换掉了
                    h.cancel_goal_async()
                    return
                self._handle = h
            h.get_result_async().add_done_callback(on_result)

        self._nav.send_goal_async(goal).add_done_callback(on_accept)

    def cancel_goal(self) -> None:
        with self._lock:
            self._seq += 1
            h, self._handle = self._handle, None
        if h is not None:
            h.cancel_goal_async()

    def set_speed_limit(self, v: float | None) -> None:
        msg = SpeedLimit()
        msg.percentage = False
        msg.speed_limit = float(v) if v else 0.0
        self._speed_pub.publish(msg)

    def set_estop(self, on: bool) -> None:
        self._estop_pub.publish(Bool(data=bool(on)))
```

```python
"""

start() 在 FastAPI lifespan 里调；测试里用 install(FakeNav()) 代替，不起 ROS。

"""
import os, threading
import process_mgr, state
from config import ROS_DOMAIN_ID, TASKS
from events import EventHub
from task_runner import RunnerConfig, TaskRunner

hub = EventHub()
runner: TaskRunner | None = None

_stop = threading.Event()


class NullNav:
    """ROS 不可用时的占位：导航永远算没启动，任务都会被拒绝"""
    def nav_running(self): return False
    def mapping(self): return False
    def current_map_id(self): return None
    def loc_score(self): return None
    def pose(self): return None
    def nearest_person(self): return None
    def send_goal(self, x, y, yaw, on_done): on_done(False, "ROS 不可用")
    def cancel_goal(self): pass
    def set_speed_limit(self, v): pass
    def set_estop(self, on): pass
    def shutdown(self): pass


def install(nav_port, cfg: RunnerConfig | None = None, estopped: bool = False) -> TaskRunner:
    global runner, nav
    nav = nav_port
    runner = TaskRunner(nav_port, cfg or RunnerConfig.from_dict(TASKS), emit=hub.publish, estopped=estopped)
    nav_port.set_estop(estopped)
    return runner


def start() -> None:
    from routers import stack           # stack 也 import runtime，放这里避免循环 import
    os.environ["ROS_DOMAIN_ID"] = str(ROS_DOMAIN_ID)
    try:
        from ros_link import RosLink
        link = RosLink(nav_running=lambda: process_mgr.is_running("nav"),
                       mapping=lambda: process_mgr.is_running("slam"),
                       current_map_id=stack.current_map_id)
    except Exception as e:
        process_mgr.log("ui", f"任务层连不上 ROS（{e}），/api/v1/tasks 不可用")
        link = NullNav()
    install(link, estopped=state.load()["estop"])
    _stop.clear()
    threading.Thread(target=_ticker, daemon=True).start()


def _ticker() -> None:
    while not _stop.wait(0.5):
        try:
            runner.tick()
        except Exception as e:
            process_mgr.log("ui", f"任务层 tick 出错: {e}")


def stop() -> None:
    _stop.set()
    if runner is not None:
        runner.cancel_all("后端退出")
    if nav is not None and hasattr(nav, "shutdown"):
        nav.shutdown()


def cancel_all(reason: str) -> None:
    if runner is not None:
        runner.cancel_all(reason)


def set_estop(on: bool) -> None:
    """急停 / 解除，同时写进 state 文件"""
    runner.estop() if on else runner.release_estop()
    state.update(lambda d: d.__setitem__("estop", bool(on)))
    process_mgr.log("ui", "软件急停" if on else "解除急停")
```

```python
"""
/api/v1：地图、点位、巡航路线。map_id 是地图名，不传 = 当前导航地图；点位 / 路线的 key 是 id 或名字。

from fastapi import APIRouter
from pydantic import BaseModel, Field
import annot_store, maps, runtime, state
from api_errors import ApiError, ok
from routers import stack

router = APIRouter(prefix="/api/v1")


def map_path(map_id: str | None) -> str:
    """对外 map_id → yaml 路径（内部仍用路径找文件）；空 = 当前导航地图"""
    if map_id:
        try:
            return maps.by_map_id(map_id)["id"]
        except KeyError:
            raise ApiError("NOT_FOUND", f"地图不存在: {map_id}")
        except ValueError as e:
            raise ApiError("AMBIGUOUS", str(e))
    cur = stack.current_map()
    if cur is None:
        raise ApiError("NAV_NOT_RUNNING", "没有正在导航的地图，请指定 map_id")
    return cur


def find_point(path: str, key: str) -> dict:
    """id 或名字精确匹配；否则名字不区分大小写，唯一才算，不猜"""
    pts = annot_store.load(path)["points"]
    for p in pts:
        if p["id"] == key or p["name"] == key:
            return p
    hits = [p for p in pts if p["name"].lower() == key.lower()]
    if len(hits) == 1:
        return hits[0]
    if hits:
        raise ApiError("AMBIGUOUS", f"「{key}」匹配到多个点: " + "、".join(p["name"] for p in hits))
    raise ApiError("NOT_FOUND", f"点不存在: {key}")


def _store(fn, *args, **kw):
    try:
        return fn(*args, **kw)
    except annot_store.StoreError as e:
        raise ApiError("BAD_REQUEST", str(e))
    except KeyError as e:
        raise ApiError("NOT_FOUND", f"不存在: {e.args[0]}")


def _route_out(path: str, r: dict) -> dict:
    names = {p["id"]: p["name"] for p in annot_store.load(path)["points"]}
    return {**r, "point_names": [names.get(i, i) for i in r["points"]]}

@router.post("/points")

    x, y, yaw = req.x, req.y, req.yaw
    if req.from_current_pose:
        if path != stack.current_map():

    return ok(_store(annot_store.create_point, path, req.name, x, y, yaw, req.note))
```

```python
"""
/api/v1：任务。goto / guide / patrol，点名在当前导航地图里查（带了 map_id 就先核对是不是当前地图）。
给上层（语音、平板、云）用的最常见调用是 POST /api/v1/guide {"point": "A会议室"}。
"""
from typing import Literal
from fastapi import APIRouter, Request
from pydantic import BaseModel, Field
import annot_store, process_mgr, runtime
from api_errors import ApiError, ok
from routers.v1_data import find_point, map_path
from task_runner import TaskError

router = APIRouter(prefix="/api/v1")


class TaskReq(BaseModel):
    type: Literal["goto", "guide", "patrol"]
    points: list[str] = Field(default_factory=list, max_length=100)   # goto：依次去的点
    point: str | None = None                                           # guide：一个点（goto 也可用）
    route: str | None = None                                           # patrol：路线 id 或名字
    loop: bool | None = None                                           # patrol：不给就按路线设置
    max_speed: float | None = Field(None, gt=0, le=0.5)
    map_id: str | None = None


class GuideReq(BaseModel):
    point: str
    max_speed: float | None = Field(None, gt=0, le=0.5)
    map_id: str | None = None


def _runner():
    if runtime.runner is None:
        raise ApiError("INTERNAL", "任务层没启动")
    return runtime.runner


def _call(fn, *args, **kw):
    try:
        return fn(*args, **kw)
    except TaskError as e:
        raise ApiError(e.code, e.msg)


def _snapshot(p: dict) -> dict:
    return {"id": p["id"], "name": p["name"], "x": p["x"], "y": p["y"], "yaw": p.get("yaw", 0.0)}


def _submit(req: TaskReq, request: Request) -> dict:
    r = _runner()
    _call(r.check, req.map_id)                     # 先核对状态和地图，再查点，错误码更准
    path = map_path(None)
    loop, route_id = False, None
    if req.type == "patrol":
        if not req.route:
            raise ApiError("BAD_REQUEST", "patrol 要指定 route")
        route = annot_store.get_route(path, req.route)
        if route is None:
            raise ApiError("NOT_FOUND", f"路线不存在: {req.route}")
        by_id = {p["id"]: p for p in annot_store.load(path)["points"]}
        pts = [by_id[i] for i in route["points"] if i in by_id]
        if not pts:
            raise ApiError("BAD_REQUEST", f"路线 {route['name']} 里没有点")
        loop = route["loop"] if req.loop is None else req.loop
        route_id = route["id"]
    else:
```
